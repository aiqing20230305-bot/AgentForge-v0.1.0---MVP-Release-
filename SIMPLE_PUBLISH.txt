━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

【方法 1】在终端执行 Git 命令（最快）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

cd ~/Downloads/world-of-claudecraft
git push origin main
git push origin v0.1.0 --force

# Git 会弹出浏览器让您授权，点击允许即可


【方法 2】在浏览器手动上传（如果上面不行）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 打开：https://github.com/Summonair/world-of-claudecraft/releases/new

2. 填写 3 个框：
   ✏️ Tag: v0.1.0
   ✏️ Title: AgentForge v0.1.0 - MVP Release 🎉
   ✏️ Description: 打开下面的网页，点击"复制"按钮

3. 点击绿色"Publish release"按钮

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📋 复制 Release 描述：
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

方式 A：打开网页（推荐）
   open WEB_PUBLISH_GUIDE.html
   → 点击"复制 Release 描述"按钮

方式 B：在终端复制
   cat GITHUB_RELEASE_COPY.txt | pbcopy
   → 内容已在剪贴板
