async getAgentPerformance(options: QueryOptions) {
  return await Agent.aggregate([
    { $match: query },
    { $lookup: { from: 'tasks', ... } },
    { $addFields: { totalTasks: { $size: '$tasks' } } },
    { $sort: { successRate: -1 } }
  ])
}

// 并行查询 - 提升性能
const [totalAgents, previousAgents, totalTasks] = await Promise.all([
  Agent.countDocuments(query),
  Agent.countDocuments({ ...query, createdAt: { $lt: startDate } }),
  Task.countDocuments(query)
])
```

export const checkPermission = (permission: string) => {
  return async (req, res, next) => {
    const user = req.user
    const hasPermission = await rbacService.hasPermission(user.id, permission)
    if (!hasPermission) {
      return res.status(403).json({ error: 'Permission denied' })
    }
    next()
  }
}
```

**Frontend**: `src/hooks/usePermission.ts`
```typescript
export const usePermission = (permission: string) => {
  const { user } = useAuth()
  return hasPermission(user, permission)
}

io.on('connection', (socket) => {
  socket.on('join:team', (teamId) => {
    socket.join(`team:${teamId}`)
  })

  socket.on('agent:update', (data) => {
    io.to(`team:${data.teamId}`).emit('agent:updated', data)
  })
})
```
